Clase `Main` (AppLauncher)

Ejemplo de clase principal para iniciar el sistema. Este bloque corresponde a `Miscelanea_1`.

package com.mycompany.miscelanea_1;

import CRUD.*;
import Modelos.*;
import java.util.Scanner;

public class Miscelanea_1 {
    private static final Scanner sc = new Scanner(System.in);
    private static final productoCRUD prodDAO = new productoCRUD();
    private static final categoriaCRUD catDAO = new categoriaCRUD();
    private static final ventaCRUD ventaDAO = new ventaCRUD();

    public static void main(String[] args) {
        int opcion;
        do {
            System.out.println("\n========== SISTEMA DE GESTIÓN INTEGRAL ==========");
            System.out.println("1. GESTIONAR PRODUCTOS");
            System.out.println("2. GESTIONAR CATEGORÍAS");
            System.out.println("3. GESTIONAR VENTAS");
            System.out.println("4. SALIR");
            System.out.print("Seleccione un módulo: ");
            opcion = sc.nextInt();
            sc.nextLine();
            switch (opcion) {
                case 1:
                    menuProducto();
                    break;
                case 2:
                    menuCategoria();
                    break;
                case 3:
                    menuVenta();
                    break;
                case 4:
                    System.out.println("Saliendo...");
                    break;
                default:
                    System.out.println("Opción inválida.");
            }
        } while (opcion != 4);
    }

    // MÉTODOS AGREGADOS PARA EVITAR ERROR DE COMPILACIÓN
    private static void menuProducto() {
        System.out.println("Entrando al menú de productos...");
    }

    private static void menuCategoria() {
        System.out.println("Entrando al menú de categorías...");
    }

    private static void menuVenta() {
        System.out.println("Entrando al menú de ventas...");
    }
}
